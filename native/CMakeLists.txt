set(ONNXRUNTIME_ROOT "" CACHE PATH "Prefix containing include/onnxruntime_cxx_api.h and lib/onnxruntime")
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" PATH_SUFFIXES onnxruntime onnxruntime/core/session)
find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" "${ONNXRUNTIME_ROOT}/lib64")
if(NOT ONNXRUNTIME_INCLUDE_DIR OR NOT ONNXRUNTIME_LIBRARY)
    message(FATAL_ERROR "Set ONNXRUNTIME_ROOT to ONNX Runtime v1.23.2 built for this target; see docs/portable.md")
endif()
add_library(oneocr_portable SHARED src/engine.cpp src/image.cpp)
target_compile_definitions(oneocr_portable PRIVATE ONEOCR_BUILDING PUBLIC ONEOCR_SHARED)
target_include_directories(oneocr_portable PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE "${ONNXRUNTIME_INCLUDE_DIR}" "${PROJECT_SOURCE_DIR}")
target_link_libraries(oneocr_portable PRIVATE "${ONNXRUNTIME_LIBRARY}")
if(MSVC)
    target_compile_options(oneocr_portable PRIVATE /W4 /utf-8 /EHsc)
else()
    target_compile_options(oneocr_portable PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(ocr src/cli.cpp)
target_include_directories(ocr PRIVATE "${PROJECT_SOURCE_DIR}")
target_link_libraries(ocr PRIVATE oneocr_portable)
if(MSVC)
    target_compile_options(ocr PRIVATE /utf-8 /EHsc)
endif()
if(APPLE)
    set_target_properties(ocr oneocr_portable PROPERTIES INSTALL_RPATH "@loader_path;@loader_path/../lib")
elseif(UNIX)
    set_target_properties(ocr oneocr_portable PROPERTIES INSTALL_RPATH "$ORIGIN;$ORIGIN/../lib")
endif()
if(WIN32)
    find_file(ORT_RUNTIME_DLL onnxruntime.dll HINTS "${ONNXRUNTIME_ROOT}/lib" "${ONNXRUNTIME_ROOT}/bin")
    if(ORT_RUNTIME_DLL)
        add_custom_command(TARGET oneocr_portable POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ORT_RUNTIME_DLL}" "$<TARGET_FILE_DIR:oneocr_portable>")
        install(FILES "${ORT_RUNTIME_DLL}" DESTINATION bin)
    endif()
endif()
if(UNIX)
    file(GLOB ORT_RUNTIME_SHARED "${ONNXRUNTIME_ROOT}/lib/libonnxruntime.so*" "${ONNXRUNTIME_ROOT}/lib/libonnxruntime*.dylib")
    if(ORT_RUNTIME_SHARED)
        install(FILES ${ORT_RUNTIME_SHARED} DESTINATION lib)
    endif()
endif()
install(TARGETS ocr oneocr_portable RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES include/oneocr_portable.h include/oneocr_portable.hpp DESTINATION include)
if(WIN32 AND NOT ONEOCR_BUILD_LEGACY)
    add_library(oneocr_wrapper_compat SHARED src/windows_compat.cpp)
    target_link_libraries(oneocr_wrapper_compat PRIVATE oneocr_portable)
    set_target_properties(oneocr_wrapper_compat PROPERTIES OUTPUT_NAME oneocr_wrapper)
    install(TARGETS oneocr_wrapper_compat RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
    install(FILES "${PROJECT_SOURCE_DIR}/oneocr.h" DESTINATION include)
endif()
if(BUILD_TESTING)
    add_executable(oneocr_api_test tests/api_test.cpp)
    target_link_libraries(oneocr_api_test PRIVATE oneocr_portable)
    add_test(NAME portable_api COMMAND oneocr_api_test)
    add_executable(oneocr_math_test tests/math_test.cpp src/image.cpp)
    target_include_directories(oneocr_math_test PRIVATE src "${PROJECT_SOURCE_DIR}")
    add_test(NAME portable_math COMMAND oneocr_math_test)
endif()
option(ONEOCR_BUILD_JNI "Build the Android JNI example" ${ANDROID})
if(ONEOCR_BUILD_JNI)
    add_library(oneocr_jni SHARED android/oneocr_jni.cpp)
    target_link_libraries(oneocr_jni PRIVATE oneocr_portable)
endif()
